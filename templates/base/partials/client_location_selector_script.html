// Public Philippine Standard Geographic Code (PSGC) data, limited to Davao Region XI. // Stored values remain "Province, City/Municipality, Barangay" for API compatibility. const LOCATION_API_BASE_URL = "https://psgc.cloud/api/v2"; const DAVAO_REGION_XI_CODE = "1100000000"; const LOCATION_REQUEST_TIMEOUT_MS = 10000; const locationApiCache = new Map(); const LOCATION_STAGE = { province: "province", city: "city", barangay: "barangay", }; const LOCATION_ACTIONS = { backProvince: "__location_back_province", backCity: "__location_back_city", currentLocation: "__location_current", retry: "__location_retry", }; const normalizeLocationItem = (item) => ({ code: String(item && item.code ? item.code : "").trim(), name: String(item && item.name ? item.name : "").trim(), }); const sortLocationItems = (items) => items .map(normalizeLocationItem) .filter((item) => item.code && item.name) .sort((first, second) => first.name.localeCompare(second.name)); const fetchLocationData = async (path) => { if (locationApiCache.has(path)) return locationApiCache.get(path); const controller = new AbortController(); const timeoutId = window.setTimeout(() => controller.abort(), LOCATION_REQUEST_TIMEOUT_MS); try { const response = await fetch(`${LOCATION_API_BASE_URL}${path}`, { headers: { Accept: "application/json" }, cache: "force-cache", signal: controller.signal, }); if (!response.ok) throw new Error(`Location service returned ${response.status}.`); const payload = await response.json(); if (!payload || !Array.isArray(payload.data)) { throw new Error("Location service returned an unexpected response."); } const data = sortLocationItems(payload.data); locationApiCache.set(path, data); return data; } finally { window.clearTimeout(timeoutId); } }; const resetSelectOptions = (selectElement, placeholder) => { if (!selectElement) return; selectElement.innerHTML = ""; const option = document.createElement("option"); option.value = ""; option.textContent = placeholder; option.disabled = true; option.selected = true; selectElement.appendChild(option); }; const appendSelectOption = (selectElement, value, label) => { if (!selectElement) return; const option = document.createElement("option"); option.value = value; option.textContent = label; selectElement.appendChild(option); }; const createLocationState = ({ hiddenInput, select, summary }) => ({ hiddenInput, select, summary, stage: LOCATION_STAGE.province, selectedProvince: null, selectedCity: null, selectedBarangay: null, provinces: [], cities: [], barangays: [], legacyValue: "", isLoading: false, errorMessage: "", requestId: 0, isBound: false, }); const getLocationSummaryParts = (state) => state.legacyValue ? [state.legacyValue] : [ state.selectedProvince && state.selectedProvince.name, state.selectedCity && state.selectedCity.name, state.selectedBarangay && state.selectedBarangay.name, ].filter(Boolean); const renderLocationSummary = (state) => { if (!state || !state.summary) return; if (state.errorMessage) { state.summary.textContent = "Location data could not be loaded. Please retry."; state.summary.dataset.locationStatus = "error"; state.summary.hidden = false; return; } const parts = getLocationSummaryParts(state); if (!parts.length) { state.summary.textContent = ""; delete state.summary.dataset.locationStatus; state.summary.hidden = true; return; } state.summary.textContent = parts.join(", "); delete state.summary.dataset.locationStatus; state.summary.hidden = false; }; const updateLocationHiddenValue = (state) => { if (!state || !state.hiddenInput) return; if (state.legacyValue) { state.hiddenInput.value = state.legacyValue; return; } const parts = [ state.selectedProvince && state.selectedProvince.name, state.selectedCity && state.selectedCity.name, state.selectedBarangay && state.selectedBarangay.name, ].filter(Boolean); state.hiddenInput.value = parts.length === 3 ? parts.join(", ") : ""; }; const getStageConfiguration = (state) => { if (state.stage === LOCATION_STAGE.province) { return { items: state.provinces, placeholder: "Select province", ariaLabel: "Select province", backAction: "", backLabel: "", selected: state.selectedProvince }; } if (state.stage === LOCATION_STAGE.city) { return { items: state.cities, placeholder: "Select city/municipality", ariaLabel: "Select city or municipality", backAction: LOCATION_ACTIONS.backProvince, backLabel: "Change province", selected: state.selectedCity }; } return { items: state.barangays, placeholder: "Select barangay", ariaLabel: "Select barangay", backAction: LOCATION_ACTIONS.backCity, backLabel: "Change city/municipality", selected: state.selectedBarangay }; }; const renderLocationSelect = (state) => { if (!state || !state.select) return; if (state.isLoading) { resetSelectOptions(state.select, "Loading locations..."); if (state.legacyValue) { appendSelectOption(state.select, LOCATION_ACTIONS.currentLocation, `Current: ${state.legacyValue}`); state.select.value = LOCATION_ACTIONS.currentLocation; } // Keep the required control enabled so a partial selection cannot bypass form validation. state.select.disabled = false; state.select.setAttribute("aria-busy", "true"); return; } const configuration = getStageConfiguration(state); resetSelectOptions(state.select, state.errorMessage ? "Unable to load locations" : configuration.placeholder); state.select.disabled = false; state.select.removeAttribute("aria-busy"); if (state.errorMessage) { appendSelectOption(state.select, LOCATION_ACTIONS.retry, "Retry loading locations"); if (configuration.backAction) appendSelectOption(state.select, configuration.backAction, configuration.backLabel); } else { if (configuration.backAction) appendSelectOption(state.select, configuration.backAction, configuration.backLabel); configuration.items.forEach((item) => appendSelectOption(state.select, item.code, item.name)); if (configuration.selected && configuration.selected.code) state.select.value = configuration.selected.code; } if (state.stage === LOCATION_STAGE.province && state.legacyValue) { appendSelectOption(state.select, LOCATION_ACTIONS.currentLocation, `Current: ${state.legacyValue}`); state.select.value = LOCATION_ACTIONS.currentLocation; } state.select.setAttribute("aria-label", configuration.ariaLabel); }; const getLocationPathForStage = (state) => { if (state.stage === LOCATION_STAGE.province) { return `/regions/${DAVAO_REGION_XI_CODE}/provinces`; } if (state.stage === LOCATION_STAGE.city && state.selectedProvince) { return `/regions/${DAVAO_REGION_XI_CODE}/provinces/${encodeURIComponent(state.selectedProvince.code)}/cities-municipalities`; } if (state.stage === LOCATION_STAGE.barangay && state.selectedProvince && state.selectedCity) { return `/regions/${DAVAO_REGION_XI_CODE}/provinces/${encodeURIComponent(state.selectedProvince.code)}/cities-municipalities/${encodeURIComponent(state.selectedCity.code)}/barangays`; } return ""; }; const setStageItems = (state, items) => { if (state.stage === LOCATION_STAGE.province) { state.provinces = items; } else if (state.stage === LOCATION_STAGE.city) { state.cities = items; } else { state.barangays = items; } }; const updateLocationValidity = (state) => { if (!state || !state.select) return; const hasLocation = Boolean(state.hiddenInput && state.hiddenInput.value.trim()); state.select.setCustomValidity(hasLocation ? "" : "Select a complete Davao Region XI location."); }; const loadLocationStage = async (state) => { if (!state) return; const path = getLocationPathForStage(state); if (!path) { state.errorMessage = "Location selection is incomplete."; state.isLoading = false; renderLocationSelect(state); renderLocationSummary(state); return; } const requestId = ++state.requestId; state.isLoading = true; state.errorMessage = ""; updateLocationHiddenValue(state); updateLocationValidity(state); renderLocationSelect(state); renderLocationSummary(state); try { const items = await fetchLocationData(path); if (requestId !== state.requestId) return; setStageItems(state, items); state.errorMessage = ""; } catch (error) { if (requestId !== state.requestId) return; state.errorMessage = error && error.name === "AbortError" ? "The location service timed out." : "The location service is unavailable."; } finally { if (requestId === state.requestId) { state.isLoading = false; renderLocationSelect(state); renderLocationSummary(state); updateLocationHiddenValue(state); updateLocationValidity(state); } } }; const selectItemByCode = (items, code) => items.find((item) => item.code === code) || null; const handleLocationSelectChange = (state) => { if (!state || !state.select || state.isLoading) return; const value = state.select.value; if (value === LOCATION_ACTIONS.currentLocation) { syncLocationState(state); return; } if (value === LOCATION_ACTIONS.retry) { loadLocationStage(state); return; } if (value === LOCATION_ACTIONS.backProvince) { state.stage = LOCATION_STAGE.province; state.selectedProvince = null; state.selectedCity = null; state.selectedBarangay = null; loadLocationStage(state); return; } if (value === LOCATION_ACTIONS.backCity) { state.stage = LOCATION_STAGE.city; state.selectedCity = null; state.selectedBarangay = null; loadLocationStage(state); return; } if (state.stage === LOCATION_STAGE.province) { const province = selectItemByCode(state.provinces, value); if (!province) return; state.selectedProvince = province; state.selectedCity = null; state.selectedBarangay = null; state.stage = LOCATION_STAGE.city; loadLocationStage(state); return; } if (state.stage === LOCATION_STAGE.city) { const city = selectItemByCode(state.cities, value); if (!city) return; state.selectedCity = city; state.selectedBarangay = null; state.stage = LOCATION_STAGE.barangay; loadLocationStage(state); return; } const barangay = selectItemByCode(state.barangays, value); if (!barangay) return; state.selectedBarangay = barangay; syncLocationState(state); }; const resetLocationState = (state) => { if (!state) return; state.stage = LOCATION_STAGE.province; state.selectedProvince = null; state.selectedCity = null; state.selectedBarangay = null; state.legacyValue = ""; state.errorMessage = ""; loadLocationStage(state); }; const bindLocationControls = (state) => { if (!state || !state.select || !state.hiddenInput) return; state.select.required = true; if (!state.isBound) { state.select.addEventListener("change", () => handleLocationSelectChange(state)); state.isBound = true; } resetLocationState(state); }; const applyLocationSelection = (state, locationValue) => { if (!state) return; const rawLocation = String(locationValue || "").trim(); if (!rawLocation) { resetLocationState(state); return; } // Preserve older or renamed locations unless the user deliberately chooses a new API value. state.stage = LOCATION_STAGE.province; state.selectedProvince = null; state.selectedCity = null; state.selectedBarangay = null; state.legacyValue = rawLocation; state.errorMessage = ""; updateLocationHiddenValue(state); updateLocationValidity(state); renderLocationSelect(state); renderLocationSummary(state); if (!state.provinces.length && !state.isLoading) loadLocationStage(state); }; const syncLocationState = (state) => { if (!state) return; updateLocationHiddenValue(state); updateLocationValidity(state); renderLocationSummary(state); };
