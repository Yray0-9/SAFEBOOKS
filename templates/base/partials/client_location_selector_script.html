const LOCATION_API_BASE_URL = "https://psgc.cloud/api/v2"; const DAVAO_REGION_CODE = "1100000000"; const DAVAO_REGION_PATH = `/regions/${DAVAO_REGION_CODE}`; const locationApiCache = new Map(); let locationCatalog; const fetchLocationData = (path) => { if (!locationApiCache.has(path)) { const controller = new AbortController(); const timeout = window.setTimeout(() => controller.abort(), 15000); const request = fetch(LOCATION_API_BASE_URL + path, { headers: { Accept: "application/json" }, cache: "force-cache", signal: controller.signal, }).then(async (response) => { if (!response.ok) throw new Error("Location service unavailable"); const payload = await response.json(); if (!Array.isArray(payload.data)) throw new Error("Invalid location response"); return payload.data; }).catch((error) => { locationApiCache.delete(path); throw error; }).finally(() => window.clearTimeout(timeout)); locationApiCache.set(path, request); } return locationApiCache.get(path); }; const locationText = (value) => String(value ?? "").trim(); const locationSearchText = (value) => locationText(value).normalize("NFD") .replace(/\p{M}/gu, "").toLowerCase().replace(/[^\p{L}\p{N}]+/gu, " ").trim(); const locationKey = (...names) => names.map(locationSearchText).join("|"); const loadLocationCatalog = async () => { const [regions, provinces, cities, barangays] = await Promise.all([ fetchLocationData("/regions"), fetchLocationData(DAVAO_REGION_PATH + "/provinces"), fetchLocationData(DAVAO_REGION_PATH + "/cities-municipalities"), fetchLocationData(DAVAO_REGION_PATH + "/barangays").catch(() => null), ]); const region = regions.find((item) => item.code === DAVAO_REGION_CODE); if (!region) throw new Error("Davao Region is unavailable"); const provinceByName = new Map(provinces.map((province) => [ locationKey(province.region, province.name), province, ])); const cityByCode = new Map(cities.map((city) => [city.code, city])); const cityByName = new Map(cities.map((city) => [ locationKey(city.region, city.province, city.name), city, ])); const result = (item, level, region, province, city, barangay) => { const names = [barangay?.name, city?.name, province?.name, region?.name] .map(locationText).filter(Boolean); // Keep the legacy most-general-first order for the saved location string. const value = [province?.name || region?.name, city?.name, barangay?.name] .map(locationText).filter(Boolean).join(", "); return { code: item.code, name: locationText(item.name), level, region, province, city, barangay, value, hierarchy: names.slice(1).join(" → "), nameKey: locationSearchText(item.name), searchKey: locationSearchText(names.join(" ") + " " + level), }; }; const results = [ result(region, "Region", region), ...provinces.map((province) => result(province, "Province", region, province)), ...cities.map((city) => result( city, city.type === "Mun" ? "Municipality" : "City", region, provinceByName.get(locationKey(region.name, city.province)), city )), ]; for (const barangay of barangays ?? []) { const city = cityByName.get(locationKey(region.name, barangay.province, barangay.city_municipality)) || cityByCode.get(barangay.code.slice(0, 7) + "000"); const province = provinceByName.get(locationKey(region.name, barangay.province)); results.push(result( barangay, "Barangay", region, province, city || { name: barangay.city_municipality }, barangay )); } return { results, incomplete: barangays === null }; }; const searchLocations = async (query) => { locationCatalog ??= loadLocationCatalog().catch((error) => { locationCatalog = null; throw error; }); const request = locationCatalog; const { results, incomplete } = await request; if (incomplete && locationCatalog === request) locationCatalog = null; const words = query.split(" "); const rank = (item) => item.nameKey === query ? 0 : item.nameKey.startsWith(query) ? 1 : words.every((word) => item.nameKey.includes(word)) ? 2 : 3; const items = results .filter((item) => words.every((word) => item.searchKey.includes(word))) .sort((a, b) => rank(a) - rank(b) || a.name.localeCompare(b.name) || a.hierarchy.localeCompare(b.hierarchy)) .slice(0, 20); return { items, incomplete }; }; const createLocationState = ({ hiddenInput, select, summary }) => ({ hiddenInput, select, summary, results: document.getElementById(select.id + "Results"), selection: null, confirmedValue: "", requestId: 0, timer: null, active: -1, }); const showLocationStatus = (state, message = "", error = false) => { state.summary.textContent = message; state.summary.hidden = !message; if (error) state.summary.dataset.locationStatus = "error"; else delete state.summary.dataset.locationStatus; }; const closeLocationResults = (state) => { window.clearTimeout(state.timer); state.requestId++; state.active = -1; state.results.replaceChildren(); state.results.hidden = true; state.select.setAttribute("aria-expanded", "false"); state.select.removeAttribute("aria-activedescendant"); state.select.removeAttribute("aria-busy"); showLocationStatus(state); }; const syncLocationState = (state) => { const confirmed = state.select.value === state.confirmedValue; state.hiddenInput.value = confirmed ? state.confirmedValue : ""; state.select.setCustomValidity(state.hiddenInput.value ? "" : "Choose a location from the search results."); }; const applyLocationSelection = (state, value) => { closeLocationResults(state); state.selection = null; state.confirmedValue = locationText(value); state.select.value = state.confirmedValue; syncLocationState(state); }; const resetLocationState = (state) => applyLocationSelection(state, ""); const chooseLocation = (state, item) => { closeLocationResults(state); state.selection = item; state.confirmedValue = item.value; state.select.value = item.value; syncLocationState(state); state.select.dispatchEvent(new Event("change", { bubbles: true })); }; const renderLocationResults = (state, items) => { items.forEach((item, index) => { const option = document.createElement("div"); option.className = "list-group-item list-group-item-action"; option.id = state.results.id + "-" + index; option.setAttribute("role", "option"); option.setAttribute("aria-selected", "false"); const name = document.createElement("strong"); name.textContent = item.name + " · " + item.level; const hierarchy = document.createElement("span"); hierarchy.className = "d-block small"; hierarchy.textContent = item.hierarchy; option.append(name, hierarchy); option.addEventListener("pointerdown", (event) => event.preventDefault()); option.addEventListener("click", () => chooseLocation(state, item)); state.results.append(option); }); state.results.hidden = !items.length; state.select.setAttribute("aria-expanded", String(Boolean(items.length))); }; const queueLocationSearch = (state) => { closeLocationResults(state); syncLocationState(state); if (state.hiddenInput.value) return; const query = locationSearchText(state.select.value); if (query.length < 2) { if (query) showLocationStatus(state, "Enter at least two characters."); return; } const requestId = state.requestId; state.select.setAttribute("aria-busy", "true"); showLocationStatus(state, "Searching locations…"); state.timer = window.setTimeout(async () => { try { const { items, incomplete } = await searchLocations(query); if (requestId !== state.requestId) return; renderLocationResults(state, items); showLocationStatus(state, incomplete ? "Some locations could not load. Press Enter to retry." : items.length ? "" : "No matching locations.", incomplete); } catch { if (requestId === state.requestId) { showLocationStatus(state, "Could not load locations. Press Enter to retry.", true); } } finally { if (requestId === state.requestId) state.select.removeAttribute("aria-busy"); } }, 250); }; const bindLocationControls = (state) => { state.select.addEventListener("input", () => queueLocationSearch(state)); state.select.addEventListener("focus", () => queueLocationSearch(state)); state.select.addEventListener("blur", () => closeLocationResults(state)); state.select.addEventListener("keydown", (event) => { const options = state.results.children; if (event.key === "Escape" && state.select.value !== state.confirmedValue) { event.preventDefault(); event.stopPropagation(); closeLocationResults(state); state.select.value = state.confirmedValue; syncLocationState(state); } else if (event.key === "Enter" && !state.hiddenInput.value) { event.preventDefault(); if (state.active >= 0) options[state.active].click(); else if (state.summary.dataset.locationStatus === "error") queueLocationSearch(state); } else if (options.length && ["ArrowDown", "ArrowUp"].includes(event.key)) { event.preventDefault(); const previous = state.active; state.active = event.key === "ArrowDown" ? Math.min(previous + 1, options.length - 1) : previous < 0 ? options.length - 1 : Math.max(previous - 1, 0); const option = options[state.active]; if (previous >= 0) { options[previous].classList.remove("active"); options[previous].setAttribute("aria-selected", "false"); } option.classList.add("active"); option.setAttribute("aria-selected", "true"); state.select.setAttribute("aria-activedescendant", option.id); option.scrollIntoView({ block: "nearest" }); } }); resetLocationState(state); };
